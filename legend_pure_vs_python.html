        constraint("TestConstraintChild", lambda this: (2 < this.i) & this.b),

simpleEpochToDate

date literal, floor, adjust, division
Pure
Mapping simpleEpochToDate
(
  FromEpochDate : Pure
  {
    ~src EpochDate
    epochDate : $src.epochDate,
    d : %1970-01-01T00:00:00.0000000
          ->adjust(floor($src.epochDate / 1000), DurationUnit.MICROSECONDS)
  }
)
Python DSL 
Mapping("simpleEpochToDate", [
    pure_set(FromEpochDate, src=EpochDate, rules=lambda s: {
        FromEpochDate.epochDate: s.epochDate,
        FromEpochDate.d: pure.date("1970-01-01T00:00:00.0000000")
            .adjust(pure.floor(s.epochDate / 1000), DurationUnit.MICROSECONDS),
    })])

simpleDataMapping

map/lambda, arithmetic, !, cast, + 3 EnumerationMappings
Pure
Mapping simpleDataMapping
(
   SomeData : Pure
   {
      ~src _SomeData
      i  : $src.i->map(n| $n * $n),
      f  : 1.0 / $src.f,
      d  : $src.d->map(n| $n + $n),
      sd : $src.sd->map(d| $d->adjust(1, DurationUnit.DAYS))->cast(@StrictDate),
      dt : $src.dt->map(t| $t->adjust(2, DurationUnit.HOURS))->cast(@DateTime),
      b  : $src.b->map(b| !$b),
      c  : EnumerationMapping ColourMapping:       $src.c,
      c2 : EnumerationMapping ColourMappingInt:    $src.i,
      c3 : EnumerationMapping ColourMappingString: $src.s
   }

   Colour : EnumerationMapping ColourMapping
   { RED : _Colour.ROUGE, BLUE : _Colour.BLEU, GREEN : _Colour.VERT }

   Colour : EnumerationMapping ColourMappingInt
   { RED : 1, BLUE : 2, GREEN : 3 }

   Colour : EnumerationMapping ColourMappingString
   { RED : 'ROUGE', BLUE : 'BLEU', GREEN : 'VERT' }
)
Python DSL 
Mapping("simpleDataMapping", [
    pure_set(SomeData, src=_SomeData, rules=lambda s: {
        SomeData.i:  s.i.map(lambda n: n * n),
        SomeData.f:  1.0 / s.f,
        SomeData.d:  s.d.map(lambda n: n + n),
        SomeData.sd: s.sd.map(lambda d: d.adjust(1, DurationUnit.DAYS)).cast(StrictDate),
        SomeData.dt: s.dt.map(lambda t: t.adjust(2, DurationUnit.HOURS)).cast(DateTime),
        SomeData.b:  s.b.map(lambda b: ~b),
        SomeData.c:  via_enum("ColourMapping",       s.c),
        SomeData.c2: via_enum("ColourMappingInt",    s.i),
        SomeData.c3: via_enum("ColourMappingString", s.s),
    }),
    enum_mapping(Colour, "ColourMapping",
                 {Colour.RED: _Colour.ROUGE, Colour.BLUE: _Colour.BLEU, Colour.GREEN: _Colour.VERT}),
    enum_mapping(Colour, "ColourMappingInt",
                 {Colour.RED: 1, Colour.BLUE: 2, Colour.GREEN: 3}),
    enum_mapping(Colour, "ColourMappingString",
                 {Colour.RED: "ROUGE", Colour.BLUE: "BLEU", Colour.GREEN: "VERT"}),
])

simpleModelMappingWithMultipleSubType

3 Person set-impls unioned over (A,B); Friendship props target ids
Pure
Mapping simpleModelMappingWithMultipleSubType
(
   Friendship : Pure
   {
      ~src _S_Friendship
      friendA[A] : $src.friendA->cast(@_S_PersonA),
      friendB[B] : $src.friendB->cast(@_S_PersonA),
      friendC[C] : if($src.friendC.friendC->isNotEmpty()
                      && $src.friendC.friendC->toOne()->instanceOf(_S_PersonB),
                      | $src.friendC.friendC->cast(@_S_PersonB),
                      | [])
   }

   Person[A] : Pure
   {
      ~src _S_PersonA
      firstName : $src.fullName->substring(0, $src.fullName->indexOf(' ')) + $src.aName,
      lastName  : $src.fullName->substring($src.fullName->indexOf(' ') + 1, $src.fullName->length())
   }
   Person[B] : Pure { ~src _S_PersonA  ... (same body, aName) }
   Person[C] : Pure { ~src _S_PersonB  ... (same body, bName) }

   *Person[union] : Operation
   {
      union_OperationSetImplementation_1__SetImplementation_MANY_(A, B)
   }
)
Python DSL 
def _person_rules(name_prop):
    return lambda s: {
        Person.firstName: s.fullName.substring(0, s.fullName.indexOf(" ")) + getattr(s, name_prop),
        Person.lastName:  s.fullName.substring(s.fullName.indexOf(" ") + 1, s.fullName.length()),
    }
pA = pure_set(Person, id="A", src=_S_PersonA, rules=_person_rules("aName"))
pB = pure_set(Person, id="B", src=_S_PersonA, rules=_person_rules("aName"))
pC = pure_set(Person, id="C", src=_S_PersonB, rules=_person_rules("bName"))
pU = union_set(Person, id="union", members=[pA, pB], root=True)

friendship = pure_set(Friendship, src=_S_Friendship, rules=lambda s: {
    Friendship.friendA["A"]: s.friendA.cast(_S_PersonA),
    Friendship.friendB["B"]: s.friendB.cast(_S_PersonA),
    Friendship.friendC["C"]: pure.if_(
        s.friendC.friendC.isNotEmpty()
            & s.friendC.friendC.toOne().instanceOf(_S_PersonB),
        s.friendC.friendC.cast(_S_PersonB),
        []),
})
Mapping("simpleModelMappingWithMultipleSubType",
        [friendship, pA, pB, pC, pU])

M

the canonical union: id-scoped target props + ~filter + Operation
Pure
Mapping M
(
  *UnionTarget : Pure
  {
    ~src UnionSource
    c[c1] : $src.a,
    c[c2] : $src.b
  }

  *UnionC[c_value] : Operation
  {
    union_OperationSetImplementation_1__SetImplementation_MANY_(c1, c2)
  }

  UnionC[c1] : Pure
  {
    ~src UnionA
    ~filter $src.fld->isNotEmpty()
    fld : $src.fld + '_a'
  }

  UnionC[c2] : Pure
  {
    ~src UnionB
    fld : $src.fld + '_b'
  }
)
Python DSL 
ut  = pure_set(UnionTarget, src=UnionSource, root=True,
               rules=lambda s: {UnionTarget.c["c1"]: s.a,
                                UnionTarget.c["c2"]: s.b})
uc1 = pure_set(UnionC, id="c1", src=UnionA,
               filter=lambda s: s.fld.isNotEmpty(),
               rules=lambda s: {UnionC.fld: s.fld + "_a"})
uc2 = pure_set(UnionC, id="c2", src=UnionB,
               rules=lambda s: {UnionC.fld: s.fld + "_b"})
ucU = union_set(UnionC, id="c_value", members=[uc1, uc2], root=True)

Mapping("M", [ut, ucU, uc1, uc2])
